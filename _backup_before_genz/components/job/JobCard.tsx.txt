import Link from "next/link";
import { MapPin, IndianRupee, MessageCircle, BadgeCheck } from "lucide-react";

// Colored org type config — UX 3
const ORG_TYPE_CONFIG: Record<string, { icon: string; bg: string; color: string }> = {
  HOTEL:    { icon: "🏨", bg: "var(--warning-light)",  color: "var(--warning)"  },
  HOSPITAL: { icon: "🏥", bg: "var(--error-light)",    color: "var(--error)"    },
  SCHOOL:   { icon: "🏫", bg: "var(--info-light)",     color: "var(--info)"     },
  SHOP:     { icon: "🏪", bg: "var(--success-light)",  color: "var(--success)"  },
  FACTORY:  { icon: "🏭", bg: "var(--surface-alt)",    color: "var(--text-secondary)" },
  OFFICE:   { icon: "🏢", bg: "var(--primary-light)",  color: "var(--primary)"  },
  HOME:     { icon: "🏠", bg: "var(--accent-light)",   color: "var(--accent)"   },
  OTHER:    { icon: "💼", bg: "var(--primary-light)",  color: "var(--primary)"  },
};

const JOB_TYPE_LABELS: Record<string, string> = {
  FULL_TIME:  "Full Time",
  PART_TIME:  "Part Time",
  CONTRACT:   "Contract",
  DAILY_WAGE: "Daily Wage",
};

const JOB_TYPE_COLORS: Record<string, { bg: string; color: string }> = {
  FULL_TIME:  { bg: "var(--info-light)",    color: "var(--info)"     },
  PART_TIME:  { bg: "var(--accent-light)",  color: "var(--accent)"   },
  CONTRACT:   { bg: "var(--warning-light)", color: "var(--warning)"  },
  DAILY_WAGE: { bg: "var(--primary-light)", color: "var(--primary)"  },
};

interface Props {
  job: {
    id: string;
    title: string;
    organization: string;
    orgType: string;
    jobType: string;
    salaryMin?: number | null;
    salaryMax?: number | null;
    salaryUnit: string;
    locality: string;
    city: string;
    contactPhone: string;
    contactWhatsapp?: string | null;
    isVerified: boolean;
    isFeatured: boolean;
  };
}

export function JobCard({ job: j }: Props) {
  const phone = j.contactWhatsapp || j.contactPhone;
  const waMsg = `Hi, I saw the job "${j.title}" at ${j.organization} on Your Mithila. I am interested.`;
  const orgCfg = ORG_TYPE_CONFIG[j.orgType] ?? ORG_TYPE_CONFIG.OTHER;
  const jobTypeCfg = JOB_TYPE_COLORS[j.jobType] ?? { bg: "var(--surface-alt)", color: "var(--text-secondary)" };

  return (
    <div className="card-warm rounded-2xl p-4">
      <div className="flex items-start gap-3">
        {/* Colored org type icon */}
        <div
          className="w-11 h-11 rounded-xl flex items-center justify-center text-xl flex-shrink-0"
          style={{ background: orgCfg.bg }}
        >
          {orgCfg.icon}
        </div>

        <div className="flex-1 min-w-0">
          <div className="flex items-start justify-between gap-2">
            <h3 className="font-semibold text-sm leading-tight" style={{ color: "var(--text-primary)" }}>
              {j.title}
            </h3>
            <span
              className="text-xs font-medium px-2 py-0.5 rounded-full whitespace-nowrap flex-shrink-0"
              style={{ background: jobTypeCfg.bg, color: jobTypeCfg.color }}
            >
              {JOB_TYPE_LABELS[j.jobType] ?? j.jobType.replace("_", " ")}
            </span>
          </div>

          <p className="text-xs mt-0.5 flex items-center gap-1" style={{ color: "var(--text-tertiary)" }}>
            {j.organization}
            {j.isVerified && (
              <BadgeCheck className="w-3 h-3 ml-0.5" style={{ color: "var(--info)" }} />
            )}
          </p>

          <div className="flex items-center gap-3 mt-2 flex-wrap">
            <span className="text-xs flex items-center gap-1" style={{ color: "var(--text-tertiary)" }}>
              <MapPin className="w-3 h-3" />
              {j.locality}
            </span>
            {j.salaryMin && (
              <span className="text-xs font-semibold flex items-center gap-0.5" style={{ color: "var(--success)" }}>
                <IndianRupee className="w-3 h-3" />
                {j.salaryMin.toLocaleString("en-IN")}
                {j.salaryMax ? `–${j.salaryMax.toLocaleString("en-IN")}` : "+"}
                <span className="font-normal" style={{ color: "var(--text-muted)" }}>
                  /{j.salaryUnit === "DAILY" ? "day" : "mo"}
                </span>
              </span>
            )}
          </div>
        </div>
      </div>

      <div className="flex gap-2 mt-4">
        <Link
          href={`/jobs/${j.id}`}
          className="flex-1 text-center text-xs font-semibold py-2.5 rounded-xl transition-colors"
          style={{ background: "var(--surface-alt)", color: "var(--text-secondary)" }}
        >
          View Details
        </Link>
        <a
          href={`https://wa.me/91${phone.replace(/\D/g, "")}?text=${encodeURIComponent(waMsg)}`}
          target="_blank"
          rel="noopener noreferrer"
          className="flex-1 text-center text-xs font-semibold py-2.5 rounded-xl transition-colors flex items-center justify-center gap-1 text-white"
          style={{ background: "#25D366" }}
        >
          <MessageCircle className="w-3.5 h-3.5" /> Apply
        </a>
      </div>
    </div>
  );
}
