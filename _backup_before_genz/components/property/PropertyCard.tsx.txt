"use client";
import Link from "next/link";
import { MapPin, BadgeCheck, Star, MessageCircle, Bed, Maximize2 } from "lucide-react";
import { SaveButton } from "@/components/property/SaveButton";

const TYPE_EMOJI: Record<string, string> = {
  FLAT: "🏢", ROOM: "🛏️", HOUSE: "🏠", SHOP: "🏪",
  LAND: "🌿", WAREHOUSE: "🏭", OFFICE: "💼", VACANT_SPACE: "📦",
};

interface Props {
  property: {
    id: string;
    title: string;
    type: string;
    purpose: string;
    price: number;
    priceUnit: string;
    locality: string;
    bedrooms?: number | null;
    area?: number | null;
    images: string[];
    contactPhone: string;
    contactWhatsapp?: string | null;
    isVerified: boolean;
    isFeatured: boolean;
  };
  savedIds?: string[]; // pass from server to hydrate heart state
}

export function PropertyCard({ property: p, savedIds = [] }: Props) {
  const phone = p.contactWhatsapp || p.contactPhone;
  const waMsg = `Hi, I saw your ${p.type} listing on Your Mithila - "${p.title}". I am interested.`;

  return (
    <div className="card-warm group rounded-2xl overflow-hidden">
      {/* Image */}
      <Link href={`/properties/${p.id}`}>
        <div className="relative h-52 overflow-hidden" style={{ background: "var(--surface-alt)" }}>
          {p.images?.[0] ? (
            <img
              src={p.images[0]}
              alt={p.title}
              className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
            />
          ) : (
            <div className="w-full h-full flex items-center justify-center text-5xl">
              {TYPE_EMOJI[p.type] ?? "🏠"}
            </div>
          )}

          {/* Featured badge */}
          {p.isFeatured && (
            <div className="absolute top-3 left-3 text-white text-xs font-bold px-2.5 py-1 rounded-full flex items-center gap-1"
              style={{ background: "var(--brand-gold, #F59E0B)" }}>
              <Star className="w-3 h-3 fill-current" /> Featured
            </div>
          )}

          {/* Purpose badge */}
          <div className="absolute top-3 right-12">
            <span className="text-xs font-semibold px-2.5 py-1 rounded-full text-white"
              style={{ background: p.purpose === "RENT" ? "var(--secondary)" : "var(--primary)" }}>
              {p.purpose === "RENT" ? "Rent" : "Sale"}
            </span>
          </div>

          {/* Save / Wishlist heart — top right */}
          <div className="absolute top-2.5 right-2.5">
            <SaveButton
              propertyId={p.id}
              initialSaved={savedIds.includes(p.id)}
              size="sm"
            />
          </div>
        </div>
      </Link>

      {/* Card body */}
      <div className="p-3">
        <p className="text-xs flex items-center gap-1 mb-0.5" style={{ color: "var(--text-tertiary)" }}>
          <MapPin className="w-3 h-3" /> {p.locality}, Mithila Region
          {p.isVerified && (
            <BadgeCheck className="w-3.5 h-3.5 ml-1" style={{ color: "var(--info)" }} />
          )}
        </p>

        <Link href={`/properties/${p.id}`}>
          <h3 className="font-semibold text-sm truncate hover:text-[var(--primary)] transition-colors"
            style={{ color: "var(--text-primary)" }}>
            {p.title}
          </h3>
        </Link>

        <div className="flex items-center gap-3 mt-0.5 text-xs" style={{ color: "var(--text-muted)" }}>
          {p.bedrooms && (
            <span className="flex items-center gap-1">
              <Bed className="w-3 h-3" />{p.bedrooms} BHK
            </span>
          )}
          {p.area && (
            <span className="flex items-center gap-1">
              <Maximize2 className="w-3 h-3" />{p.area} sq ft
            </span>
          )}
        </div>

        <div className="flex items-center justify-between mt-2">
          <p className="font-bold text-sm" style={{ color: "var(--primary)" }}>
            ₹{p.price.toLocaleString("en-IN")}
            <span className="font-normal text-xs" style={{ color: "var(--text-tertiary)" }}>
              {p.priceUnit === "MONTHLY" ? "/mo" : p.priceUnit === "YEARLY" ? "/yr" : ""}
            </span>
          </p>
          <a
            href={`https://wa.me/91${phone.replace(/\D/g, "")}?text=${encodeURIComponent(waMsg)}`}
            target="_blank"
            rel="noopener noreferrer"
            className="flex items-center gap-1 text-xs font-medium px-2.5 py-1 rounded-full transition-colors text-white"
            style={{ background: "#25D366" }}
            onClick={e => e.stopPropagation()}
          >
            <MessageCircle className="w-3 h-3" /> WhatsApp
          </a>
        </div>
      </div>
    </div>
  );
}
