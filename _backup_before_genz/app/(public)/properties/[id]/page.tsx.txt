import { prisma } from "@/lib/prisma";
import { notFound } from "next/navigation";
import type { Metadata } from "next";
import Image from "next/image";
import Link from "next/link";
import {
  MapPin, Bed, Bath, Maximize2, Building2, BadgeCheck,
  Star, Phone, Calendar, MessageCircle, ArrowLeft,
} from "lucide-react";
import { MapEmbed } from "@/components/shared/MapEmbed";
import { StickyContactBar } from "@/components/shared/StickyContactBar";
import { formatPrice, PROPERTY_TYPE_LABELS } from "@/lib/utils";
import { PropertyReviews } from "@/components/property/PropertyReviews";
import { WriteReviewForm } from "@/components/property/WriteReviewForm";
import { EMICalculator } from "@/components/property/EMICalculator";
import { auth } from "@/lib/auth";

export async function generateMetadata({ params }: { params: { id: string } }): Promise<Metadata> {
  try {
    const property = await prisma.property.findUnique({
      where: { id: params.id },
      select: {
        title: true, type: true, purpose: true, price: true,
        locality: true, city: true, description: true, images: true,
      },
    });
    if (!property) return { title: "Property Not Found | YourMithila" };

    const typeLabel =
      property.type === "FLAT"      ? "Flat"
      : property.type === "ROOM"    ? "Room"
      : property.type === "HOUSE"   ? "House"
      : property.type === "SHOP"    ? "Shop"
      : property.type === "LAND"    ? "Land"
      : property.type === "OFFICE"  ? "Office"
      : (PROPERTY_TYPE_LABELS[property.type] ?? property.type.replace("_", " "));

    const purposeLabel = property.purpose === "RENT" ? "for Rent" : "for Sale";

    return {
      title: `${property.title} | ${typeLabel} ${purposeLabel} in ${property.locality}, Madhubani`,
      description: `${typeLabel} ${purposeLabel} in ${property.locality}, ${property.city}, Bihar. ₹${property.price.toLocaleString("en-IN")}. ${property.description?.substring(0, 120) ?? ""}. Contact owner directly on WhatsApp.`,
      keywords: [
        `${typeLabel.toLowerCase()} ${purposeLabel} ${property.locality}`,
        `${typeLabel.toLowerCase()} ${purposeLabel} madhubani`,
        `property in ${property.locality}`,
        `property in madhubani`,
        `${property.locality} ${property.city} property`,
      ],
      openGraph: {
        title: `${property.title} | YourMithila`,
        description: `${typeLabel} ${purposeLabel} in ${property.locality}, Madhubani Bihar. ₹${property.price.toLocaleString("en-IN")}`,
        images:
          property.images && property.images.length > 0
            ? [{ url: property.images[0], width: 800, height: 600, alt: property.title }]
            : [{ url: "/og-image.png", width: 1200, height: 630, alt: "YourMithila" }],
        url: `https://yourmithila.in/properties/${params.id}`,
        type: "website",
      },
      alternates: {
        canonical: `https://yourmithila.in/properties/${params.id}`,
      },
    };
  } catch {
    return { title: "Property | YourMithila" };
  }
}

const CATEGORY_EMOJI: Record<string, string> = {
  FLAT: "🏢", ROOM: "🛏️", HOUSE: "🏠", SHOP: "🏪",
  LAND: "🌿", WAREHOUSE: "🏭", OFFICE: "💼", VACANT_SPACE: "📐",
};

export default async function PropertyDetailPage({ params }: { params: { id: string } }) {
  let property;
  try {
    property = await prisma.property.findUnique({
      where: { id: params.id },
      include: { owner: { select: { name: true, phone: true, createdAt: true } } },
    });
  } catch (error) {
    console.error("Property fetch error:", error);
    return notFound();
  }

  if (!property) return notFound();

  // Increment view count (non-blocking)
  prisma.property.update({
    where: { id: params.id },
    data: { viewCount: { increment: 1 } },
  }).catch(console.error);

  // Fetch similar properties — same locality first, then same city
  const similarProperties = await prisma.property.findMany({
    where: {
      id:          { not: property.id },
      isAvailable: true,
      OR: [
        { locality: property.locality },
        { city:     property.city     },
      ],
    },
    take: 3,
    orderBy: [{ isFeatured: "desc" }, { createdAt: "desc" }],
  }).catch(() => []);

  // Fetch approved reviews for this property
  const reviews = await prisma.review.findMany({
    where: { isApproved: true, propertyId: params.id },
    orderBy: { createdAt: "desc" },
    take: 10,
  }).catch(() => []);

  // Check if user is logged in (for review form)
  const session = await auth();

  const phone = property.contactWhatsapp || property.contactPhone;
  const waMessage = `Hi, I saw your ${PROPERTY_TYPE_LABELS[property.type] ?? property.type} listing on Your Mithila - "${property.title}". I am interested.`;
  const waUrl = `https://wa.me/91${phone.replace(/\D/g, "")}?text=${encodeURIComponent(waMessage)}`;

  return (
    <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-8 pb-24 md:pb-8">
      {/* Breadcrumb */}
      <div className="flex items-center gap-2 text-sm mb-6" style={{ color: "var(--text-tertiary)" }}>
        <Link href="/" className="hover:text-[var(--primary)] transition-colors">Home</Link>
        <span>/</span>
        <Link href="/properties" className="hover:text-[var(--primary)] transition-colors">Properties</Link>
        <span>/</span>
        <span style={{ color: "var(--text-primary)" }} className="truncate max-w-[200px]">{property.title}</span>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
        {/* Main content */}
        <div className="lg:col-span-2 space-y-6">
          {/* Image gallery */}
          <div className="space-y-2">
            <div className="relative h-72 sm:h-96 rounded-2xl overflow-hidden" style={{ background: "var(--surface-alt)" }}>
              {property.images[0] ? (
                <Image src={property.images[0]} alt={property.title} fill className="object-cover" priority />
              ) : (
                <div className="w-full h-full flex items-center justify-center text-6xl">
                  {CATEGORY_EMOJI[property.type] ?? "🏠"}
                </div>
              )}
              {property.isFeatured && (
                <div className="absolute top-3 left-3 text-white text-xs font-bold px-2.5 py-1 rounded-full flex items-center gap-1"
                  style={{ background: "var(--brand-gold, #F59E0B)" }}>
                  <Star className="w-3 h-3 fill-current" /> Featured
                </div>
              )}
              <div className="absolute top-3 right-3">
                <span className={`text-xs font-semibold px-2.5 py-1 rounded-full text-white`}
                  style={{ background: property.purpose === "RENT" ? "var(--secondary)" : "var(--primary)" }}>
                  {property.purpose === "RENT" ? "For Rent" : "For Sale"}
                </span>
              </div>
            </div>
            {property.images.length > 1 && (
              <div className="grid grid-cols-4 gap-2">
                {property.images.slice(1, 5).map((img, i) => (
                  <div key={i} className="relative h-20 rounded-xl overflow-hidden" style={{ background: "var(--surface-alt)" }}>
                    <Image src={img} alt={`Photo ${i + 2}`} fill className="object-cover" />
                  </div>
                ))}
              </div>
            )}
          </div>

          {/* Title & badges */}
          <div>
            <div className="flex items-start gap-2 flex-wrap">
              <span className="text-xs font-semibold px-2.5 py-1 rounded-full"
                style={{ background: "var(--primary-light)", color: "var(--primary)" }}>
                {PROPERTY_TYPE_LABELS[property.type] ?? property.type}
              </span>
              {property.isVerified && (
                <span className="flex items-center gap-1 text-xs font-semibold px-2.5 py-1 rounded-full"
                  style={{ background: "var(--success-light)", color: "var(--success)" }}>
                  <BadgeCheck className="w-3 h-3" /> Verified
                </span>
              )}
            </div>
            <h1 className="text-2xl font-bold mt-3" style={{ color: "var(--text-primary)" }}>{property.title}</h1>
            <div className="flex items-center gap-1 text-sm mt-1" style={{ color: "var(--text-tertiary)" }}>
              <MapPin className="w-4 h-4" />
              {property.locality}, {property.city}, {property.district}, {property.state}
              {property.pincode && ` — ${property.pincode}`}
            </div>
          </div>

          {/* Price */}
          <div className="rounded-xl p-4" style={{ background: "var(--primary-light)" }}>
            <p className="text-3xl font-black" style={{ color: "var(--primary)" }}>
              {formatPrice(property.price, property.priceUnit)}
            </p>
            {property.area && (
              <p className="text-sm mt-1" style={{ color: "var(--text-tertiary)" }}>
                ₹{Math.round(property.price / property.area).toLocaleString("en-IN")}/sq ft
              </p>
            )}
          </div>

          {/* Details grid */}
          {(property.bedrooms || property.bathrooms || property.area || property.floor) && (
            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
              {property.bedrooms && (
                <div className="rounded-xl p-3 text-center" style={{ background: "var(--surface-alt)" }}>
                  <Bed className="w-5 h-5 mx-auto mb-1" style={{ color: "var(--text-tertiary)" }} />
                  <p className="font-bold" style={{ color: "var(--text-primary)" }}>{property.bedrooms}</p>
                  <p className="text-xs" style={{ color: "var(--text-tertiary)" }}>Bedrooms</p>
                </div>
              )}
              {property.bathrooms && (
                <div className="rounded-xl p-3 text-center" style={{ background: "var(--surface-alt)" }}>
                  <Bath className="w-5 h-5 mx-auto mb-1" style={{ color: "var(--text-tertiary)" }} />
                  <p className="font-bold" style={{ color: "var(--text-primary)" }}>{property.bathrooms}</p>
                  <p className="text-xs" style={{ color: "var(--text-tertiary)" }}>Bathrooms</p>
                </div>
              )}
              {property.area && (
                <div className="rounded-xl p-3 text-center" style={{ background: "var(--surface-alt)" }}>
                  <Maximize2 className="w-5 h-5 mx-auto mb-1" style={{ color: "var(--text-tertiary)" }} />
                  <p className="font-bold" style={{ color: "var(--text-primary)" }}>{property.area}</p>
                  <p className="text-xs" style={{ color: "var(--text-tertiary)" }}>Sq Ft</p>
                </div>
              )}
              {property.floor && (
                <div className="rounded-xl p-3 text-center" style={{ background: "var(--surface-alt)" }}>
                  <Building2 className="w-5 h-5 mx-auto mb-1" style={{ color: "var(--text-tertiary)" }} />
                  <p className="font-bold" style={{ color: "var(--text-primary)" }}>
                    {property.floor}{property.totalFloors ? `/${property.totalFloors}` : ""}
                  </p>
                  <p className="text-xs" style={{ color: "var(--text-tertiary)" }}>Floor</p>
                </div>
              )}
            </div>
          )}

          {/* Description */}
          <div>
            <h2 className="font-bold mb-2" style={{ color: "var(--text-primary)" }}>Description</h2>
            <p className="text-sm leading-relaxed whitespace-pre-line" style={{ color: "var(--text-secondary)" }}>
              {property.description}
            </p>
          </div>

          {/* Amenities */}
          {property.amenities.length > 0 && (
            <div>
              <h2 className="font-bold mb-2" style={{ color: "var(--text-primary)" }}>Amenities</h2>
              <div className="flex flex-wrap gap-2">
                {property.amenities.map((a) => (
                  <span key={a} className="text-xs font-medium px-3 py-1 rounded-full border"
                    style={{ background: "var(--success-light)", color: "var(--success)", borderColor: "var(--success)" }}>
                    ✓ {a}
                  </span>
                ))}
              </div>
            </div>
          )}

          {/* Map */}
          <div>
            <h2 className="font-bold mb-2" style={{ color: "var(--text-primary)" }}>Location</h2>
            <MapEmbed address={property.address} latitude={property.latitude} longitude={property.longitude} />
          </div>

          {/* Reviews section */}
          <PropertyReviews reviews={reviews} />

          {/* Write review form */}
          {session ? (
            <WriteReviewForm propertyId={property.id} />
          ) : (
            <div className="mt-4 text-center py-4 rounded-xl"
              style={{ background: "var(--surface-alt)", border: "1px solid var(--border)" }}>
              <p className="text-sm" style={{ color: "var(--text-tertiary)" }}>
                <Link href="/auth/login" className="font-medium hover:underline" style={{ color: "var(--primary)" }}>
                  Login
                </Link>{" "}
                to write a review
              </p>
            </div>
          )}
        </div>

        {/* Sidebar */}
        <div className="space-y-4">
          {/* Contact card */}
          <div className="rounded-2xl p-5 shadow-sm sticky top-20" style={{ background: "var(--surface)", border: "1px solid var(--border)" }}>
            <h3 className="font-bold mb-4" style={{ color: "var(--text-primary)" }}>Contact Owner</h3>
            <div className="space-y-3">
              <div>
                <p className="text-xs" style={{ color: "var(--text-tertiary)" }}>Contact Person</p>
                <p className="font-semibold" style={{ color: "var(--text-primary)" }}>{property.contactName}</p>
              </div>
              <a href={`tel:${property.contactPhone}`}
                className="flex items-center gap-2 w-full font-medium py-3 px-4 rounded-xl transition-colors text-sm"
                style={{ background: "var(--surface-alt)", color: "var(--text-primary)" }}>
                <Phone className="w-4 h-4" /> {property.contactPhone}
              </a>
              <a href={waUrl} target="_blank" rel="noopener noreferrer"
                className="flex items-center justify-center gap-2 w-full text-white font-semibold py-3 px-4 rounded-xl transition-colors text-sm"
                style={{ background: "#25D366" }}>
                <MessageCircle className="w-4 h-4" /> WhatsApp Owner
              </a>
            </div>

            <div className="mt-4 pt-4 space-y-1 text-xs" style={{ borderTop: "1px solid var(--border)", color: "var(--text-tertiary)" }}>
              <div className="flex items-center gap-1">
                <Calendar className="w-3 h-3" />
                Listed {new Date(property.createdAt).toLocaleDateString("en-IN")}
              </div>
              <p>👁 {property.viewCount} views</p>
            </div>
          </div>

          {/* EMI Calculator — only for SALE properties */}
          {property.purpose === "SALE" && (
            <EMICalculator price={property.price} />
          )}
        </div>
      </div>

      {/* Similar Properties */}
      {similarProperties.length > 0 && (
        <div className="mt-12">
          <h2 className="text-xl font-bold mb-4" style={{ color: "var(--text-primary)" }}>
            Similar Properties
          </h2>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
            {similarProperties.map((p) => {
              const pPhone = p.contactWhatsapp || p.contactPhone;
              const pWaMsg = `Hi, I saw your ${PROPERTY_TYPE_LABELS[p.type] ?? p.type} listing on Your Mithila - "${p.title}". I am interested.`;
              return (
                <div key={p.id} className="card-warm rounded-2xl overflow-hidden">
                  <Link href={`/properties/${p.id}`}>
                    <div className="relative h-40 overflow-hidden" style={{ background: "var(--surface-alt)" }}>
                      {p.images?.[0] ? (
                        <Image src={p.images[0]} alt={p.title} fill className="object-cover hover:scale-105 transition-transform duration-300" />
                      ) : (
                        <div className="w-full h-full flex items-center justify-center text-4xl">
                          {CATEGORY_EMOJI[p.type] ?? "🏠"}
                        </div>
                      )}
                    </div>
                  </Link>
                  <div className="p-3">
                    <Link href={`/properties/${p.id}`}>
                      <h3 className="font-semibold text-sm truncate hover:text-[var(--primary)] transition-colors" style={{ color: "var(--text-primary)" }}>
                        {p.title}
                      </h3>
                    </Link>
                    <p className="text-xs mt-0.5 flex items-center gap-1" style={{ color: "var(--text-tertiary)" }}>
                      <MapPin className="w-3 h-3" /> {p.locality}
                    </p>
                    <div className="flex items-center justify-between mt-2">
                      <p className="font-bold text-sm" style={{ color: "var(--primary)" }}>
                        ₹{p.price.toLocaleString("en-IN")}
                      </p>
                      <a href={`https://wa.me/91${pPhone.replace(/\D/g, "")}?text=${encodeURIComponent(pWaMsg)}`}
                        target="_blank" rel="noopener noreferrer"
                        className="text-xs font-medium px-2.5 py-1 rounded-full text-white"
                        style={{ background: "#25D366" }}>
                        WhatsApp
                      </a>
                    </div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      )}

      {/* Sticky mobile contact bar */}
      <StickyContactBar
        phone={property.contactPhone}
        waPhone={phone}
        waMessage={waMessage}
        label="WhatsApp Owner"
      />
    </div>
  );
}
