function PropertyCard({ p }: { p: any }) {
  const phone = p.contactWhatsapp || p.contactPhone;
  const waMsg = `Hi, I saw your ${PROPERTY_TYPE_LABELS[p.type] ?? p.type} listing on Your Mithila - "${p.title}". I am interested.`;
  return (
    <div className="card-warm rounded-2xl overflow-hidden group">
      <Link href={`/properties/${p.id}`}>
        <div className="relative h-48 overflow-hidden" style={{ background: "var(--surface-alt)" }}>
          {p.images?.[0] ? (
            <img src={p.images[0]} alt={p.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300" />
          ) : (
            <div className="w-full h-full flex items-center justify-center text-5xl">
              {CATEGORIES.find(c => c.type === p.type)?.emoji ?? "🏠"}
            </div>
          )}
          {p.isFeatured && (
            <div className="absolute top-3 left-3 text-white text-xs font-bold px-2.5 py-1 rounded-full flex items-center gap-1"
              style={{ background: "#F59E0B" }}>
              <Star className="w-3 h-3 fill-current" /> Featured
            </div>
          )}
          <div className="absolute top-3 right-3">
            <span className="text-xs font-semibold px-2.5 py-1 rounded-full text-white"
              style={{ background: p.purpose === "RENT" ? "var(--secondary)" : "var(--primary)" }}>
              {p.purpose === "RENT" ? "Rent" : "Sale"}
            </span>
          </div>
        </div>
      </Link>
      <div className="p-3">
        <p className="text-xs flex items-center gap-1 mb-0.5" style={{ color: "var(--text-tertiary)" }}>
          <MapPin className="w-3 h-3" /> {p.locality}, Mithila Region
          {p.isVerified && <BadgeCheck className="w-3.5 h-3.5 ml-1" style={{ color: "var(--info)" }} />}
        </p>
        <Link href={`/properties/${p.id}`}>
          <h3 className="font-semibold text-sm truncate hover:text-[var(--primary)] transition-colors" style={{ color: "var(--text-primary)" }}>{p.title}</h3>
        </Link>
        <div className="flex items-center justify-between mt-2">
          <p className="font-bold text-sm" style={{ color: "var(--primary)" }}>
            ₹{p.price.toLocaleString("en-IN")}
            <span className="font-normal text-xs" style={{ color: "var(--text-tertiary)" }}>
              {p.priceUnit === "MONTHLY" ? "/mo" : p.priceUnit === "YEARLY" ? "/yr" : ""}
            </span>
          </p>
          <a href={`https://wa.me/91${phone}?text=${encodeURIComponent(waMsg)}`} target="_blank" rel="noopener noreferrer"
            className="flex items-center gap-1 text-xs font-medium px-2.5 py-1 rounded-full transition-colors text-white"
            style={{ background: "#25D366" }}>
            <MessageCircle className="w-3 h-3" /> WhatsApp
          </a>
        </div>
      </div>
    </div>
  );
}

function JobCard({ j }: { j: any }) {
  const phone = j.contactWhatsapp || j.contactPhone;
  const waMsg = `Hi, I saw the job "${j.title}" at ${j.organization} on Your Mithila. I am interested.`;
  const typeColors: Record<string, string> = {
    FULL_TIME: "var(--secondary)", PART_TIME: "var(--accent)", DAILY_WAGE: "var(--primary)", CONTRACT: "#7B5EA7",
  };
  return (
    <div className="card-warm rounded-2xl p-4">
      <div className="flex items-start gap-3">
        <div className="w-11 h-11 rounded-xl flex items-center justify-center text-xl flex-shrink-0" style={{ background: "var(--primary-light)" }}>🏢</div>
        <div className="flex-1 min-w-0">
          <div className="flex items-start justify-between gap-2">
            <h3 className="font-semibold text-sm leading-tight" style={{ color: "var(--text-primary)" }}>{j.title}</h3>
            <span className="text-xs font-medium px-2 py-0.5 rounded-full whitespace-nowrap flex-shrink-0 text-white"
              style={{ background: typeColors[j.jobType] ?? "var(--text-tertiary)" }}>
              {j.jobType.replace("_", " ")}
            </span>
          </div>
          <p className="text-xs mt-0.5" style={{ color: "var(--text-tertiary)" }}>{j.organization}</p>
          <div className="flex items-center gap-3 mt-2">
            <span className="text-xs flex items-center gap-1" style={{ color: "var(--text-tertiary)" }}>
              <MapPin className="w-3 h-3" />{j.locality}
            </span>
            {j.salaryMin && (
              <span className="text-xs font-semibold flex items-center gap-0.5" style={{ color: "var(--success)" }}>
                <IndianRupee className="w-3 h-3" />{j.salaryMin.toLocaleString("en-IN")}+
              </span>
            )}
          </div>
        </div>
      </div>
      <div className="flex gap-2 mt-3">
        <Link href={`/jobs/${j.id}`} className="flex-1 text-center text-xs font-semibold py-2.5 rounded-xl transition-colors"
          style={{ background: "var(--surface-alt)", color: "var(--text-secondary)" }}>
          View Details
        </Link>
        <a href={`https://wa.me/91${phone}?text=${encodeURIComponent(waMsg)}`} target="_blank" rel="noopener noreferrer"
          className="flex-1 text-center text-xs font-semibold py-2.5 rounded-xl transition-colors flex items-center justify-center gap-1 text-white"
          style={{ background: "#25D366" }}>
          <MessageCircle className="w-3 h-3" /> Apply
        </a>
      </div>
    </div>
  );
}
